#include <tunables/global>

# Complain mode: AppArmor logs what it would block instead of blocking it.
# Remove `complain` once a full school day logs no ALLOWED entries.
profile myride_bridge flags=(attach_disconnected,mediate_deleted,complain) {
  #include <abstractions/base>
  #include <abstractions/nameservice>
  #include <abstractions/ssl_certs>

  signal (send) set=(kill,term,int,hup,cont),
  signal (receive),

  # Docker's init (init: true), the bridge, and the image HEALTHCHECK (wget)
  /sbin/docker-init ix,
  /usr/local/bin/node ix,
  /bin/busybox ix,
  /bin/sh ix,

  # The bridge's code, system configuration, and runtime introspection
  /app/** r,
  /etc/** r,
  /usr/share/zoneinfo/** r,
  @{PROC}/** r,
  /sys/fs/cgroup/** r,
  /sys/devices/system/cpu/** r,

  # options.json and the persisted refresh token
  /data/ r,
  /data/** rw,
  /tmp/** rwk,

  # Cognito, MyRide SignalR, MQTT, and the Supervisor API
  network inet stream,
  network inet6 stream,
  network inet dgram,
  network inet6 dgram,
  network netlink raw,
}
